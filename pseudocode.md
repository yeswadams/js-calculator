START PROGRAM
CREATE AN EMPTY ARRAY FOR THE HISTORY OF CALCULATIONS

    FUNCTION ADD(NUMBER1, NUMBER2)
        RETURN NUMBER1 + NUMBER2
    END FUNCTION

    FUNCTION SUBTRACT(NUMBER1, NUMBER2)
        RETURN NUMBER1 - NUMBER2
    END FUNCTION

    FUNCTION MULTIPLY(NUMBER1, NUMBER2)
        RETURN NUMBER1 * NUMBER2
    END FUNCTION

    FUNCTION DIVIDE(NUMBER1, NUMBER2)
        IF NUMBER2 == 0 THEN OR
            RETURN "Error: Cannot divide by zero"
        ELSE
        RETURN NUMBER1 / NUMBER2
    ENDIF
    END FUNCTION

    FUNCTION SAVETOHISTORY(OPERATION, OPERANDS[NUMBER1, NUMBER2], RESULT)
        ADD THE OPERATION, OPERANDS, AND RESULT TO THE HISTORY ARRAY .PUSH()
        RETURN THE HISTORY
    END FUNCTION



END PROGRAM
